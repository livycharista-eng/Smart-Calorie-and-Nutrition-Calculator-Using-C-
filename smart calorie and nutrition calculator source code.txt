#include <iostream>
#include <iomanip>

using namespace std;

int main() {
    // Basic Input Variables
    int gender;
    double weight, height;
    int age;
    int activityOption;
    double activityFactor = 0.0;

    cout << "========================================" << endl;
    cout << "   NUTRITION & TDEE CALCULATOR          " << endl;
    cout << "========================================" << endl;

    // 1. Gender Input with Validation Loop
    while (true) {
        cout << "Enter Gender (1 = male, 2 = female): ";
        if (cin >> gender && (gender == 1 || gender == 2)) {
            break; // Valid input
        } else {
            cout << "[Error] Invalid input! Please enter 1 or 2.\n" << endl;
            cin.clear();
            cin.ignore(1000, '\n');
        }
    }

    // 2. Weight, Height, and Age Input with Validation Loop
    while (true) {
        cout << "Enter Weight (kg): ";
        cin >> weight;
        cout << "Enter Height (cm): ";
        cin >> height;
        cout << "Enter Age (years): ";
        cin >> age;

        if (weight > 0 && height > 0 && age > 0) {
            break; // Valid input
        } else {
            cout << "[Error] Weight, height, and age must be positive values!\n" << endl;
            cin.clear();
            cin.ignore(1000, '\n');
        }
    }

    // 3. Daily Activity Factor Input
    while (true) {
        cout << "\nSelect Daily Activity Factor (1-5):" << endl;
        cout << "1. Little to no exercise (1.2)" << endl;
        cout << "2. Light Activity / 1-3 days/week (1.375)" << endl;
        cout << "3. Moderate Activity / 3-5 days/week (1.55)" << endl;
        cout << "4. Heavy Activity / 6-7 days/week (1.725)" << endl;
        cout << "5. Very Heavy Activity / Athlete (1.9)" << endl;
        cout << "Your Choice (1-5): ";

        if (cin >> activityOption && activityOption >= 1 && activityOption <= 5) {
            switch (activityOption) {
                case 1: activityFactor = 1.2; break;
                case 2: activityFactor = 1.375; break;
                case 3: activityFactor = 1.55; break;
                case 4: activityFactor = 1.725; break;
                case 5: activityFactor = 1.9; break;
            }
            break; // Valid input
        } else {
            cout << "[Error] Invalid activity factor choice! Please enter a number between 1 and 5.\n" << endl;
            cin.clear();
            cin.ignore(1000, '\n');
        }
    }

    // 4. Calculate BMR
    double bmr = 0.0;
    if (gender == 1) { // Male
        bmr = (10.0 * weight) + (6.25 * height) - (5.0 * age) + 5.0;
    } else { // Female
        bmr = (10.0 * weight) + (6.25 * height) - (5.0 * age) - 161.0;
    }

    // 5. Calculate TDEE
    double tdee = bmr * activityFactor;

    // 6. Calculate Macronutrient Targets (Following Flowchart & do-while rules)
    double targetProtein = 0.0;
    double targetCarbs = 0.0;
    double targetFat = 0.0;
    bool isInputSuccess = false;

    do {
        double remainingPersen = 100.0;

        cout << "\n----------------------------------------" << endl;
        cout << "   MACRONUTRIENT TARGET SETTING (%)     " << endl;
        cout << "----------------------------------------" << endl;

        // Input Target Protein (10% - 35%)
        cout << "Enter Target Protein (10% - 35%): ";
        cin >> targetProtein;

        if (targetProtein < 10.0 || targetProtein > 35.0) {
            cout << "[Error] Protein must be in the range of 10% - 35%!\n";
            continue; // Retry from the start of do-while loop
        }
        remainingPersen -= targetProtein;

        // Input Target Carbs (45% - 65%)
        cout << "Enter Target Carbs (45% - 65%): ";
        cin >> targetCarbs;

        if (targetCarbs < 45.0 || targetCarbs > 65.0) {
            cout << "[Error] Carbs must be in the range of 45% - 65%!\n";
            continue; // Retry from the start of do-while loop
        }
        remainingPersen -= targetCarbs;

        // Target Fat calculation from remaining percentage
        targetFat = remainingPersen;

        // Check if targetFat fits the required range (20% - 35%)
        if (targetFat >= 20.0 && targetFat <= 35.0) {
            isInputSuccess = true;
        } else {
            cout << "[Error] Remaining percentage for Fat is " << targetFat 
                 << "%. Fat target must be between 20% and 35%!\n";
            cout << "The total percentage must equal exactly 100%. Please re-enter the target percentages.\n";
        }

    } while (!isInputSuccess);

    // 7. Convert Macronutrients to Grams
    double proteinGrams = (tdee * (targetProtein / 100.0)) / 4.0;
    double carbsGrams   = (tdee * (targetCarbs / 100.0)) / 4.0;
    double fatGrams     = (tdee * (targetFat / 100.0)) / 9.0;

    // 8. Output Results
    cout << fixed << setprecision(0);
    cout << "\n========================================" << endl;
    cout << "            CALCULATION RESULTS         " << endl;
    cout << "========================================" << endl;
    cout << "BMR  : " << bmr << " kcal/day" << endl;
    cout << "TDEE : " << tdee << " kcal/day" << endl;
    cout << "----------------------------------------" << endl;
    cout << "Macronutrient Targets:" << endl;
    cout << "- Protein    (" << targetProtein << "%) : " << proteinGrams << " grams" << endl;
    cout << "- Carbs      (" << targetCarbs << "%) : " << carbsGrams << " grams" << endl;
    cout << "- Fat        (" << targetFat << "%) : " << fatGrams << " grams" << endl;
    cout << "========================================" << endl;

    return 0;
}
